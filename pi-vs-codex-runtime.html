PI · AGENTSESSION IN YOUR PROCESS 
import {
  createAgentSession,
  SessionManager,
} from "@earendil-works/pi-coding-agent";

const { session } = await createAgentSession({
  cwd: repo,
  tools: [
    "read", "grep", "find", "ls",
    "bash", "edit", "write"
  ],
  sessionManager: SessionManager.inMemory(repo),
});

session.subscribe((event) => {
  if (
    event.type === "message_update" &&
    event.assistantMessageEvent.type === "text_delta"
  ) {
    process.stdout.write(
      event.assistantMessageEvent.delta
    );
  }
});

await session.prompt(`
  修复 auth.test.ts，先复现，
  做最小修改并运行相关测试。
`);

session.dispose();
CODEX · SDK SPAWNS NATIVE BINARY 
import { Codex } from "@openai/codex-sdk";

const codex = new Codex();

const thread = codex.startThread({
  workingDirectory: repo,
  sandboxMode: "workspace-write",
  approvalPolicy: "on-request",
});

const { events } = await thread.runStreamed(`
  修复 auth.test.ts，先复现，
  做最小修改并运行相关测试。
`);

for await (const event of events) {
  if (event.type === "item.completed") {
    console.log(event.item);
  }

  if (event.type === "turn.completed") {
    console.log(event.usage);
  }
}

你得到可变对象。Session、messages、tools 与生命周期 callback 都在同一进程。

你得到事件流。真正的 tool router、compaction、approval 与 sandbox 在 Codex 进程里。

PI · POLICY HOOK 
export default function security(pi) {
  const secret = /(^|\/)\.env(?:\.|$)/;

  pi.on("tool_call", async (event) => {
    if (["read", "write", "edit"]
        .includes(event.toolName)) {
      const path = String(event.input.path ?? "");

      if (secret.test(path)) {
        return {
          block: true,
          reason: `禁止访问：${path}`,
        };
      }
    }

    if (event.toolName === "bash") {
      const cmd = String(
        event.input.command ?? ""
      );

      if (/\bgit\s+push\b/.test(cmd)) {
        return {
          block: true,
          reason: "禁止 agent push",
        };
      }
    }
  });
}
CODEX · PERMISSION PROFILE 
# .codex/config.toml
default_permissions = "repo-safe"

[permissions.repo-safe]
description = "仓库可写，环境文件不可读"
extends = ":workspace"

[permissions.repo-safe.filesystem]
glob_scan_max_depth = 3

[permissions.repo-safe.filesystem.":workspace_roots"]
"**/*.env" = "deny"

[[hooks.PreToolUse]]
matcher = "^Bash$"

[[hooks.PreToolUse.hooks]]
type = "command"
command = """
python3 "$(git rev-parse --show-toplevel)\
/.codex/hooks/no_push.py"
"""
timeout = 5

注意：这是策略拦截。`python open(".env")` 仍可能绕过路径字符串检查；强隔离需容器或替换执行 backend。

文件规则是权限边界。在 sandbox 生效时，Python、Node、Ruby 间接读取同样会被拒绝。
